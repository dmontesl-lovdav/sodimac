CREATE PROCEDURE [dbo].[uspRangoFactura]
AS

DECLARE 
@V_RANGO          AS NUMERIC(9),
@V_RANGO_TMP      AS NUMERIC(9),
@V_RANGO_MAX_TMP  AS NUMERIC(9),
@V_RANGO_MIN_TMP  AS NUMERIC(9),

SET @V_RANGO_TMP = 100;
