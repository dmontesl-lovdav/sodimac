SELECT id, code, prefix, name, module
FROM shared_catalogs.catalog_header
WHERE code IN ('CatPerfil','CatModulo','CatAplicativo','CatEvento','CatAtributo','CatRol','CatPermiso')
ORDER BY code;
También:


SELECT h.code, COUNT(d.id) AS detalles_existentes
FROM shared_catalogs.catalog_header h
LEFT JOIN shared_catalogs.catalog_detail d ON d.header_id = h.id
WHERE h.code IN ('CatPerfil','CatModulo','CatAplicativo','CatEvento','CatAtributo','CatRol','CatPermiso')
GROUP BY h.code ORDER BY h.code;
