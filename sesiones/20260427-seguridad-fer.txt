SELECT

  ua.user_data_id,

  t.id   AS attribute_type_id,

  t.key  AS attribute_type_key,

  v.id   AS attribute_value_id,

  v.key  AS attribute_value_key

FROM core_security.user_attribute ua

JOIN shared_catalogs.catalog_detail t

  ON t.id = ua.catalog_detail_attribute_type_id

JOIN shared_catalogs.catalog_detail v

  ON v.id = ua.catalog_detail_attribute_value_id

WHERE ua.user_data_id = 2

  AND ua.status = 1
