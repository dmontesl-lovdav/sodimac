SELECT t.GRUPO, t.TICKET,
       SUBSTRING(t.TICKET,1,8) AS FECHA_NUM, SUBSTRING(t.TICKET,9,4) AS TIENDA_NUM,
       (SELECT COUNT(*) FROM dbo.VENTA_CAB v WHERE v.TICKET = t.TICKET)                  AS en_venta_cab,
       (SELECT MAX(v.TIPO) FROM dbo.VENTA_CAB v WHERE v.TICKET = t.TICKET)               AS tipo,
       (SELECT MAX(v.SUBTOTAL) FROM dbo.VENTA_CAB v WHERE v.TICKET = t.TICKET)           AS subtotal,
